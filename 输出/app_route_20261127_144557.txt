``` tsx
// src/App.tsx
import { useState, useEffect } from 'react';
import { BrowserRouter as Router, Routes, Route, Navigate, useLocation } from 'react-router-dom';
import Sidebar from './components/layout/Sidebar';
import FileCenter from './pages/FileCenter';
import AICopilot from './pages/AICopilot';
import Notes from './pages/Notes';
import ScreenShare from './pages/ScreenShare';
import UserManagement from './pages/UserManagement';
import Login from './pages/Login';
import { useAuthStore, applyThemeToDom } from './store/useAuthStore';
import { useNoteStore } from './store/useNoteStore';
import { resolveDocumentTitle } from './config/siteConfig';
import { KeyRound, X, AlertCircle, Save } from 'lucide-react';

function RouteTitleSync() {
  const location = useLocation();
  useEffect(() => {
    document.title = resolveDocumentTitle(location.pathname, true);
  }, [location.pathname]);
  return null;
}

function App() {
  const { currentUser, theme, changePassword, syncCurrentUser } = useAuthStore();
  const { unsavedGuardOpen, isSavingBeforeLeave, resolveUnsavedGuard } = useNoteStore();

  const [isPwdModalOpen, setIsPwdModalOpen] = useState(false);
  const [newPwd, setNewPwd] = useState('');
  const [confirmPwd, setConfirmPwd] = useState('');
  const [pwdError, setPwdError] = useState('');
  const [isBannerIgnored, setIsBannerIgnored] = useState(false);

  useEffect(() => {
    applyThemeToDom(theme);
  }, [theme]);

  useEffect(() => {
    if (currentUser && currentUser.role !== 'guest') {
      syncCurrentUser();
    }
    if (!currentUser) {
      document.title = resolveDocumentTitle('/login', false);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [currentUser?.id]);

  useEffect(() => {
    const handleOpen = () => setIsPwdModalOpen(true);
    window.addEventListener('open_change_pwd_modal', handleOpen);
    return () => window.removeEventListener('open_change_pwd_modal', handleOpen);
  }, []);

  if (!currentUser) {
    return <Login />;
  }

  const handleSavePassword = async () => {
    if (newPwd.length < 6) {
      setPwdError('密码不能少于 6 位');
      return;
    }
    if (newPwd !== confirmPwd) {
      setPwdError('两次输入密码不一致');
      return;
    }
    try {
      await changePassword(newPwd);
      setIsPwdModalOpen(false);
      setNewPwd('');
      setConfirmPwd('');
      setPwdError('');
      alert('密码修改成功！');
    } catch (err) {
      setPwdError(err instanceof Error ? err.message : '密码修改失败');
    }
  };

  const canUseAi = currentUser.role !== 'guest';

  return (
    <Router>
      <RouteTitleSync />
      <div className="flex h-screen w-full font-sans text-gray-200 bg-[#1e2128] overflow-hidden relative pb-14 md:pb-0">
        <Sidebar />

        <main className="flex-1 relative overflow-hidden h-full">
          <Routes>
            <Route path="/" element={<Navigate to="/file-center" replace />} />
            <Route path="/file-center/*" element={<FileCenter />} />
            <Route
              path="/ai-copilot/*"
              element={canUseAi ? <AICopilot /> : <Navigate to="/file-center" replace />}
            />
            <Route path="/notes/*" element={<Notes />} />
            <Route path="/screen-share/*" element={<ScreenShare />} />
            {currentUser.role === 'admin' && (
              <Route path="/users/*" element={<UserManagement />} />
            )}
            <Route path="*" element={<Navigate to="/file-center" replace />} />
          </Routes>
        </main>

        {/* 首次登录右下角悬浮提示卡片 */}
        {currentUser.mustChangePassword && !isBannerIgnored && (
          <div className="absolute bottom-6 right-6 bg-[#252830] border border-yellow-500/50 rounded-2xl p-4 w-80 shadow-2xl z-40">
            <div className="flex items-start justify-between">
              <div className="flex items-center gap-2 text-yellow-400 font-bold text-xs">
                <AlertCircle size={16} />
                <span>账号安全提醒</span>
              </div>
              <button
                type="button"
                onClick={() => setIsBannerIgnored(true)}
                className="text-gray-500 hover:text-white"
              >
                <X size={14} />
              </button>
            </div>
            <p className="text-[11px] text-gray-300 mt-2 mb-3 leading-relaxed">
              您当前使用的是系统分配的初始密码，为了保证数据安全，请尽快修改。
            </p>
            <div className="flex items-center justify-end gap-2">
              <button
                type="button"
                onClick={() => setIsBannerIgnored(true)}
                className="text-[11px] text-gray-400 hover:text-gray-200 px-2 py-1"
              >
                稍后提醒
              </button>
              <button
                type="button"
                onClick={() => {
                  setIsBannerIgnored(true);
                  setIsPwdModalOpen(true);
                }}
                className="text-[11px] bg-yellow-500/20 text-yellow-300 border border-yellow-500/40 px-2.5 py-1 rounded-lg hover:bg-yellow-500/30"
              >
                立即修改
              </button>
            </div>
          </div>
        )}

        {/* 修改密码弹窗 */}
        {isPwdModalOpen && (
          <div className="fixed inset-0 bg-black/60 backdrop-blur-xs flex items-center justify-center z-50 p-4">
            <div className="bg-[#252830] border border-gray-700 rounded-2xl w-full max-w-sm p-5 shadow-2xl">
              <div className="flex items-center justify-between mb-4">
                <div className="flex items-center gap-2 text-sm font-bold text-white">
                  <KeyRound size={16} className="text-blue-400" />
                  <span>修改登录密码</span>
                </div>
                <button
                  type="button"
                  onClick={() => {
                    setIsPwdModalOpen(false);
                    setPwdError('');
                  }}
                  className="text-gray-400 hover:text-white"
                >
                  <X size={16} />
                </button>
              </div>
              <div className="space-y-3">
                <div>
                  <label className="block text-xs text-gray-400 mb-1">新密码（至少 6 位）</label>
                  <input
                    type="password"
                    value={newPwd}
                    onChange={(e) => setNewPwd(e.target.value)}
                    className="w-full bg-[#1a1d24] border border-gray-700 rounded-lg px-3 py-2 text-sm text-white focus:outline-none focus:border-blue-500"
                    placeholder="输入新密码"
                  />
                </div>
                <div>
                  <label className="block text-xs text-gray-400 mb-1">确认新密码</label>
                  <input
                    type="password"
                    value={confirmPwd}
                    onChange={(e) => setConfirmPwd(e.target.value)}
                    className="w-full bg-[#1a1d24] border border-gray-700 rounded-lg px-3 py-2 text-sm text-white focus:outline-none focus:border-blue-500"
                    placeholder="再次输入新密码"
                  />
                </div>
                {pwdError && <p className="text-xs text-rose-400">{pwdError}</p>}
              </div>
              <div className="flex justify-end gap-2 mt-5">
                <button
                  type="button"
                  onClick={() => {
                    setIsPwdModalOpen(false);
                    setPwdError('');
                  }}
                  className="px-3 py-1.5 text-xs rounded-lg text-gray-400 hover:bg-gray-800"
                >
                  取消
                </button>
                <button
                  type="button"
                  onClick={handleSavePassword}
                  className="px-3.5 py-1.5 text-xs rounded-lg bg-blue-600 hover:bg-blue-500 text-white font-medium"
                >
                  保存密码
                </button>
              </div>
            </div>
          </div>
        )}

        {/* 离开未保存笔记确认弹窗 */}
        {unsavedGuardOpen && (
          <div className="fixed inset-0 bg-black/65 backdrop-blur-xs flex items-center justify-center z-50 p-4">
            <div className="bg-[#252830] border border-gray-700 rounded-2xl w-full max-w-sm p-5 shadow-2xl">
              <div className="flex items-center gap-2 text-amber-400 font-bold text-sm mb-2">
                <AlertCircle size={17} />
                <span>当前笔记有未保存修改</span>
              </div>
              <p className="text-xs text-gray-300 leading-relaxed mb-4">
                离开当前页面前是否先保存笔记修改？
              </p>
              <div className="flex items-center justify-end gap-2">
                <button
                  type="button"
                  disabled={isSavingBeforeLeave}
                  onClick={() => resolveUnsavedGuard('cancel')}
                  className="px-3 py-1.5 text-xs rounded-lg text-gray-400 hover:bg-gray-800"
                >
                  取消
                </button>
                <button
                  type="button"
                  disabled={isSavingBeforeLeave}
                  onClick={() => resolveUnsavedGuard('discard')}
                  className="px-3 py-1.5 text-xs rounded-lg bg-gray-700/70 hover:bg-gray-700 text-gray-200"
                >
                  放弃修改
                </button>
                <button
                  type="button"
                  disabled={isSavingBeforeLeave}
                  onClick={() => resolveUnsavedGuard('save')}
                  className="inline-flex items-center gap-1 px-3.5 py-1.5 text-xs rounded-lg bg-blue-600 hover:bg-blue-500 text-white font-medium"
                >
                  <Save size={13} />
                  <span>{isSavingBeforeLeave ? '保存中...' : '保存并继续'}</span>
                </button>
              </div>
            </div>
          </div>
        )}
      </div>
    </Router>
  );
}

export default App;
```
